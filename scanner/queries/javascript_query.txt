import io.shiftleft.semanticcpg.language._
import io.shiftleft.codepropertygraph.generated.nodes._

val results = cpg.file.filter(_.name == "{filename}").flatMap {{ f =>
    val allNodes = f.ast.toList
    val cleanTarget = "{snippet}".replaceAll("\\s", "")

    // 1. Find the specific sink line
    val sinkNode = allNodes
        .filter {{ n => 
            val code = n.propertyOption[String]("CODE").getOrElse("")
            !n.isInstanceOf[Method] && code.replaceAll("\\s", "").contains(cleanTarget)
        }}
        .sortBy(_.propertyOption[String]("CODE").getOrElse("").length)
        .headOption

    sinkNode.map {{ sn =>
        val sinkLine = sn.propertyOption[Any]("LINE_NUMBER").map(_.toString.toInt).getOrElse(0)
        
        // 2. Define our "Chain of Interest"
        // We want variables from the sink, plus the bridge variables
        val sinkVars = f.ast.lineNumber(sinkLine).isIdentifier.name.toSet
        val bridgeVars = Set("workspace", "path", "file", "hash_file")
        val allTracked = sinkVars ++ bridgeVars

        // 3. Trace: Only grab lines that mention our specific data chain
        val trace = f.ast
            .filter(_.propertyOption("LINE_NUMBER").isDefined)
            .filter {{ n =>
                val c = n.propertyOption[String]("CODE").getOrElse("")
                val name = n.propertyOption[String]("NAME").getOrElse("")
                
                // Only include if it's not a Method block AND:
                // a) Uses one of our specific variables
                // b) Is the call to our function
                !n.isInstanceOf[Method] && (
                    allTracked.exists(v => c.contains(v) || name == v) || 
                    c.contains("is_up_to_date")
                )
            }}
            .toList
            .groupBy(_.propertyOption[Any]("LINE_NUMBER").get.toString.toInt)
            .map {{ case (line, nodes) =>
                val bestCode = nodes.map(_.propertyOption[String]("CODE").getOrElse("")).maxByOption(_.length).getOrElse("")
                (line, s"L$line: $bestCode")
            }}
            .filter {{ case (_, code) => 
                // Final noise reduction: ignore lines that are just "this" or very short
                !code.contains(": this") && code.length > 10
            }}
            .toList
            .sortBy(_._1)
            .map(_._2)
            .distinct

        Map(
            "sink" -> java.util.Base64.getEncoder.encodeToString(sn.propertyOption[String]("CODE").getOrElse("{snippet}").getBytes),
            "full_trace" -> trace.map(t => java.util.Base64.getEncoder.encodeToString(t.getBytes))
        )
    }}
}}.toList.toJson

"###RESULT###" + results + "###RESULT###"
