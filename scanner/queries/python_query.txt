val filename = "{filename}"
val snippet = "{snippet}"

val results = cpg.file.name(".*" + filename).method.filter(_.ast.code.exists(_.contains(snippet))).map {{ m =>
    val sinkNode = m.ast.filter(_.code.contains(snippet)).headOption
    val sinkLine = sinkNode.flatMap(_.lineNumber).getOrElse(0)

    // 1. Get the names of identifiers on the sink line (e.g., "digest")
    val trackedVars = m.ast.lineNumber(sinkLine).isIdentifier.name.toSet

    // 2. Find every node in the method that CONTAINS these variable names
    
    val trace = m.ast
        .filter(n => 
            trackedVars.exists(v => n.code.contains(v)) || 
            (n.label == "METHOD_PARAMETER_IN" && trackedVars.contains(n.property("NAME").toString))
        ).l

        .groupBy(_.lineNumber.getOrElse(0))
        .map {{ case (line, nodes) =>
            // Pick the longest string on that line to get the full statement context
            val bestCode = nodes.map(_.code).maxBy(_.length)
            (line, "L" + line + ": " + bestCode)

        }}.toList
        .sortBy(_._1)
        .map(_._2)
        .distinct

    Map(

        "sink" -> java.util.Base64.getEncoder.encodeToString(sinkNode.map(_.code).getOrElse(snippet).getBytes),
        "full_trace" -> trace.map(t => java.util.Base64.getEncoder.encodeToString(t.getBytes))
    )

}}.toJson



"###RESULT###" + results + "###RESULT###"
